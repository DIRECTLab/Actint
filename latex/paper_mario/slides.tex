\begin{frame}{The Problem: Vessels Go Dark}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{90,000 vessels} of regulatory concern active at any moment.

      \bigskip
      \alert{75\% of industrial fishing vessels} intentionally disable
      AIS transponders [Paolo et al., \textit{Nature} 2024].

      \bigskip
      Even when AIS is active, analysts must classify:
      \begin{itemize}
        \item \textbf{Activity} — fishing vs.\ transit vs.\ loitering vs.\ STS
        \item \textbf{Type} — trawler vs.\ cargo vs.\ naval contact
      \end{itemize}

      \bigskip
      \textbf{Core difficulty:} custody windows range from a single
      EO frame to hours of AIS history.
    \end{column}

\begin{frame}{Feature Engineering: 42 Dimensions, Any $N \geq 1$}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Key design constraint: features must be computable for \textbf{any}
      observation count. NaN where data is insufficient — XGBoost learns
      optimal split directions for missing values.

      \medskip
      \begin{tabular}{p{0.8cm}p{4.2cm}r}
        \toprule
        Min $N$ & Feature group & Count \\
        \midrule
        1  & SOG, heading, sensor metadata,\\
           & position context, time-of-day & 17 \\
        2  & Acceleration, displacement,\\
           & AIS gap statistics & 7 \\
        4  & Speed pattern ratios & 4 \\
        6  & Net-to-gross, zig-zag score & 3 \\
        10 & Loiter index, bounding box,\\
           & mean turn rate & 4 \\
        \bottomrule
      \end{tabular}

      \medskip
      \textbf{Loiter index} = convex-hull area / path$^2$ —
      discriminates fishing from transit.
    \end{column}
    \begin{column}{0.45\textwidth}
      \includegraphics[width=\textwidth]{../outputs/brazil_eez/feature_importance.png}
      {\small\textit{Feature importance — Brazil EEZ model.
      SOG mean, nav\_status, and pct\_slow dominate.}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{XGBoost + GPU Acceleration}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Configuration:}
      \begin{itemize}
        \item 500 estimators, depth 8, lr 0.05
        \item Subsample 0.8 · colsample 0.8
        \item $\ell_2$ regularisation $\lambda = 1.0$
        \item \texttt{hist} tree method on CUDA
        \item Hardware: NVIDIA RTX 4080 Laptop (11 GB)
      \end{itemize}

      \bigskip
      \textbf{Computational performance:}
      \begin{tabular}{lc}
        \toprule
        Operation & Time \\
        \midrule
        Feature extraction (216k) & 47 s \\
        XGBoost activity training  & 38 s \\
        XGBoost type training      & 44 s \\
        Inference (64k examples)   & 1.2 s \\
        sklearn RF (equivalent)    & $\approx$14 min \\
        \bottomrule
      \end{tabular}

      \medskip
      \alert{$22\times$ speedup} over CPU-based RandomForest.\\
      \textbf{19 $\mu$s/prediction} — supports real-time deployment.
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Why XGBoost beats neural networks here:}
      \begin{itemize}
        \item Native NaN handling — no imputation needed
        \item Calibrated probability outputs
        \item SHAP interpretability built-in
        \item No minimum sequence length
      \end{itemize}

      \bigskip
      \textbf{Train/test split is vessel-based} (70/30), not ping-based.
      All windows from a given MMSI are in one partition only.

      \bigskip
      SHAP analysis: \texttt{sog\_mean} + \texttt{pct\_slow} account for
      $>60\%$ of mean absolute SHAP contribution.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Dark Vessel Detection}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Composite risk score $d \in [0,1]$:
      \[
        d = 0.60 \cdot s_\text{gap}
          + 0.30 \cdot s_\text{spoof}
          + 0.10 \cdot s_\text{flag}
      \]

      \begin{itemize}
        \item $s_\text{gap}$: number + duration of gaps $>1$ h +
              dark-ping fraction
        \item $s_\text{spoof}$: kinematic impossibilities
              (implied speed $>40$ kn)
        \item $s_\text{flag}$: flag-of-convenience registry penalty
      \end{itemize}

      \bigskip
      \textbf{Results (Dec 2023 + Jan 2024):}\\
      25 + 24 vessels flagged with $d > 0.5$.\\
      All flagged: \alert{fishing-registered} — consistent with known
      IUU pattern in regulatory boundary zones.

      \medskip
      SHAP top contributors: gap duration, n\_gaps, vessel type prior.
    \end{column}
    \begin{column}{0.45\textwidth}
      \includegraphics[width=\textwidth]{../outputs/figures/B5_risk_score_dashboard.png}
      {\small\textit{Risk score dashboard — composite IUU + dark scores
      per vessel. Red markers $>0.5$ threshold.}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Confidence Calibration}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Each prediction is accompanied by a calibrated confidence score:
      \[
        c = 0.30\,q_s
          + 0.45\!\min\!\!\left(1,\frac{\ln(N+1)}{\ln 21}\right)
          + 0.25\,\phi
      \]
      where $q_s$ = sensor quality (AIS = 1.0, EO = 0.4),
      $N$ = observation count (saturates at $\approx$20),
      $\phi$ = fraction of non-NaN features.

      \bigskip
      \textbf{Intentionally conservative:}\\
      Single EO frame → $c \approx 0.30$ regardless of model probability.

      \medskip
      This acknowledges the fundamental information deficit of a
      one-frame contact and prevents over-confident downstream tasking.
    \end{column}
    \begin{column}{0.45\textwidth}
      \includegraphics[width=\textwidth]{../outputs/figures/A7_confidence_calibration.png}
      {\small\textit{Confidence calibration curve — reliability diagram
      across observation densities.}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Discussion: The EO/SAR Generalisation Gap}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Open challenge:} single EO/SAR observations where
      nav\_status and SOG are unavailable.

      \bigskip
      Current behaviour:
      \begin{itemize}
        \item Confidence $c \approx 0.30$ for single EO frame
        \item Predicted class depends on \textbf{spatial context}
              (proximity to fishing grounds, shipping lanes, time of day)
        \item Geography prior not yet encoded as a feature
      \end{itemize}

      \bigskip
      \textbf{Paths forward:}
      \begin{itemize}
        \item SAR foundation models (SARatrX) — pre-trained on
              unlabelled target data, few-shot vessel recognition
        \item Maritime geography prior — encode known lane/zone
              membership as additional features
        \item SAR/AIS fusion layer — simultaneous kinematic + identity
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{Fishing detection in cross-dataset eval:}

      \medskip
      \begin{tabular}{lcc}
        \toprule
        Dataset & Prec. & Recall \\
        \midrule
        Jun-2023  & 1.00  & $\approx$1.00 \\
        Dec-2023  & 1.00  & 0.12 \\
        Jan-2024  & 1.00  & 0.18 \\
        \bottomrule
      \end{tabular}

      \medskip
      Zero false positives throughout.\\
      Missed cases are fishing vessels \textbf{transiting} — the ground
      truth nav\_status correctly reads 0 (under way).

      \medskip
      The right metric for IUU detection without nav\_status is
      \textbf{vessel-type} recall ($F_1 = 1.00$), not activity recall.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Limitations and Future Work}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Current limitations:}
      \begin{enumerate}
        \item \textbf{Geography:} real training data confined to US
              coastal + Eastern Mediterranean. Open-ocean performance
              extrapolated via simulation.
        \item \textbf{Temporal:} three daily snapshots — limited
              seasonal diversity for rare activity classes.
        \item \textbf{STS/loiter GT:} no real ground truth for
              STS and loitering in NOAA data; synthetic evaluation only.
        \item \textbf{Class imbalance:} fishing $<1\%$ of US coastal
              AIS traffic. Pacific/Indian Ocean data would improve recall.
      \end{enumerate}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Priority future directions:}
      \begin{itemize}
        \item Ingest real AIS from Global Fishing Watch API or
              exactEarth — pipeline is data-agnostic
        \item SAR/AIS fusion track association layer
        \item SAR foundation backbone (SARatrX) for EO-only contacts
        \item Maritime geography prior as feature layer
        \item Incorporate Pacific/Indian Ocean fishing data to address
              class imbalance
      \end{itemize}

      \bigskip
      The feature and classifier pipeline is already designed to accept
      these data sources — integration is an \textbf{engineering} task,
      not a research one.
    \end{column}
  \end{columns}
\end{frame}
